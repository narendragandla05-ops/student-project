#!/bin/bash

FILE="students.txt"

while true
do
    echo "=============================="
    echo " Student Record Management "
    echo "=============================="
    echo "1. Add Student"
    echo "2. View Students"
    echo "3. Search Student"
    echo "4. Delete Student"
    echo "5. Update Student"
    echo "6. Exit"
    echo "Enter choice:"
    read choice

    case $choice in

    1)
        echo "Enter Student ID:"
        read id

        echo "Enter Student Name:"
        read name

        echo "Enter Course:"
        read course

        echo "$id | $name | $course" >> $FILE

        echo "Student Added Successfully!"
        ;;

    2)
        echo "Student Records:"
        cat $FILE
        ;;

    3)
        echo "Enter Student ID to Search:"
        read sid

        grep "^$sid" $FILE

        if [ $? -ne 0 ]
        then
            echo "Record Not Found"
        fi
        ;;

    4)
        echo "Enter Student ID to Delete:"
        read did

        grep -v "^$did" $FILE > temp.txt
        mv temp.txt $FILE

        echo "Record Deleted"
        ;;

    5)
        echo "Enter Student ID to Update:"
        read uid

        grep -v "^$uid" $FILE > temp.txt

        echo "Enter New Name:"
        read newname

        echo "Enter New Course:"
        read newcourse

        echo "$uid | $newname | $newcourse" >> temp.txt

        mv temp.txt $FILE

        echo "Record Updated"
        ;;

    6)
        echo "Exiting Program..."
        break
        ;;

    *)
        echo "Invalid Choice"
        ;;
    esac

done
